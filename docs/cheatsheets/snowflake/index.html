Compute

CREATE WAREHOUSE wh WAREHOUSE_SIZE = MEDIUMCompute is separate from storage
AUTO_SUSPEND = 60 AUTO_RESUME = TRUEThe single biggest cost control
ALTER WAREHOUSE wh SET WAREHOUSE_SIZE = LARGEResize instantly, no data movement
MAX_CLUSTER_COUNT = 4Multi-cluster for concurrency, not for speed
USE WAREHOUSE / DATABASE / SCHEMA / ROLESession context

Time travel & cloning

SELECT * FROM t AT (OFFSET => -3600)One hour ago
SELECT * FROM t BEFORE (STATEMENT => '01a…')Before a specific statement
UNDROP TABLE tRestore within retention
CREATE TABLE clone CLONE prodZero-copy clone — instant, no storage
DATA_RETENTION_TIME_IN_DAYS = 70–90 depending on edition
