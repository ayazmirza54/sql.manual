Connecting

mysql -h 127.0.0.1 -P 3306 -u root -p db⚠ uppercase -P for port
mysql --defaults-file=~/.my.cnf dbCredentials from a file
--ssl-mode=REQUIREDEnforce TLS
localhost vs 127.0.0.1⚠ socket vs TCP — different grants

Flags

-e "SQL"Execute and exit
-B / --batchTab-separated for piping
--tableForce table output off a TTY
--verticalOne column per line
--safe-updatesRefuse keyless UPDATE/DELETE
-f / --forceContinue past errors

SHOW

SHOW DATABASES; SHOW TABLES;Navigate
SHOW CREATE TABLE t\GFull DDL
SHOW INDEX FROM t;Indexes and cardinality
SHOW FULL PROCESSLIST;What is running
SHOW ENGINE INNODB STATUS\GDeadlocks, buffer pool, transactions
SHOW VARIABLES LIKE '%timeout%';Configuration
SHOW STATUS LIKE 'Threads%';Runtime counters

Diagnostics

SELECT * FROM sys.schema_unused_indexesDead weight
SELECT * FROM sys.statements_with_full_table_scansMissing indexes
performance_schema.events_statements_summary_by_digestSlowest by total time
EXPLAIN ANALYZE SELECT …8.0.18+
KILL QUERY 42; KILL 42;Cancel query / connection

Backup & load

mysqldump --single-transaction --routines --triggers db > db.sqlNon-blocking on InnoDB
mysqldump --set-gtid-purged=ONFor replica rebuilds
mysql db < db.sqlRestore
LOAD DATA LOCAL INFILE 'f.csv' INTO TABLE tFast bulk load
xtrabackup --backupPhysical hot backup
