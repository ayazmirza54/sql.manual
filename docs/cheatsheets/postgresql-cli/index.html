Connecting

psql -h host -p 5432 -U user -d dbExplicit flags
psql "postgresql://u:p@host/db?sslmode=require"URI form
PGHOST / PGUSER / PGDATABASE / PGPASSWORDEnvironment
~/.pgpass (mode 0600)host:port:db:user:password

Flags

-c "SQL"Run one statement and exit
-f script.sqlRun a file
-v ON_ERROR_STOP=1⚠ essential for migrations
--single-transactionAll-or-nothing script
-A -t -F ','CSV-ish output for pipes
-EEcho the SQL behind meta-commands
-XIgnore ~/.psqlrc

Meta-commands

\l \c db \conninfoDatabases and connection
\dt \dt+ \d t \d+ tTables and structure
\di \dv \dm \df \dn \du \dpIndexes, views, matviews, functions, schemas, roles, privileges
\x auto \pset null '¤'Output formatting
\timing \watch 5Timing, repeat every 5 s
\e \ef funcEdit query / function in $EDITOR
\copy t FROM 'f.csv' CSV HEADERClient-side COPY
\i f.sql \o out.txt \! lsInclude, redirect, shell
\h SELECT \?SQL help, meta-command help

Catalog queries

SELECT pg_size_pretty(pg_database_size(current_database()))Database size
SELECT relname, pg_size_pretty(pg_total_relation_size(oid)) FROM pg_class ORDER BY pg_total_relation_size(oid) DESC LIMIT 10Biggest tables
SELECT * FROM pg_stat_activity WHERE state <> 'idle'Active queries
SELECT * FROM pg_stat_replicationReplica lag
SELECT * FROM pg_locks WHERE NOT grantedBlocked
SELECT pg_terminate_backend(pid)Kill a session

Backup

pg_dump -Fc -f app.dump appCustom format (compressed, selective restore)
pg_restore -d newdb -j 4 app.dumpParallel restore
pg_dump -t orders --data-only appOne table
pg_basebackup -D /backup -Ft -z -X streamPhysical base backup
pg_dumpall --globals-onlyRoles and tablespaces
