$ sql --explain "latest order per customer"

-- PostgreSQL · fastest with (customer_id, created_at DESC)
SELECT DISTINCT ON (customer_id) *
FROM   orders
ORDER BY customer_id, created_at DESC;

-- Portable · any engine with window functions
SELECT * FROM (
  SELECT o.*, ROW_NUMBER() OVER (
           PARTITION BY customer_id
           ORDER BY created_at DESC) AS rn
  FROM orders o
) t WHERE rn = 1;
